**Automatisez les réponses à vos questions juridiques et comptables avec l'IA**

## 📋 Description

L'Assistant Juridique & Comptable IA est une application qui automatise les réponses aux questions juridiques et comptables en utilisant vos propres documents. Elle repose sur une architecture RAG (Retrieval-Augmented Generation) avec indexation vectorielle.

### 🎯 Fonctionnalités

- ✅ **Indexation de documents** - PDF, textes juridiques, contrats
- ✅ **Questions en langage naturel** - Posez vos questions en français
- ✅ **Réponses sourcées** - Chaque réponse cite ses documents sources
- ✅ **Niveau de confiance** - L'IA indique quand elle n'est pas sûre
- ✅ **Fallback humain** - Questions complexes → expert
- ✅ **Authentification** - Comptes cabinets avec JWT
- ✅ **Abonnements** - Gestion Stripe (Solo, Pro, Cabinet, Enterprise)
- ✅ **Facturation** - Génération PDF de factures
- ✅ **Backups** - Sauvegarde automatique (local + cloud)
- ✅ **OCR** - Support des PDF scannés

# create_zip.py
"""
Crée le package ZIP pour GitHub
Exécutez ce script pour générer le fichier à déposer sur GitHub
"""

import os
import shutil
import zipfile
from pathlib import Path
from datetime import datetime

def create_github_package():
    """Crée le package ZIP pour GitHub"""
    
    # Dossiers à inclure
    include_dirs = [
        "backend",
        "frontend",
        "email_templates",
        "credentials",
        "scripts",
        "docs",
    ]
    
    # Fichiers à inclure à la racine
    include_files = [
        ".env.example",
        ".gitignore",
        "README.md",
        "LICENSE",
        "docker-compose.yml",
        "requirements.txt",
        "install_windows.bat",
        "start_app.bat",
        "run_backup.bat",
    ]
    
    # Dossiers à exclure (data et backups)
    exclude_dirs = ["data", "backups", "__pycache__", ".venv", "venv", ".git"]
    
    # Créer le ZIP
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    zip_name = f"assistant-juridique-ia_{timestamp}.zip"
    
    with zipfile.ZipFile(zip_name, 'w', zipfile.ZIP_DEFLATED) as zipf:
        
        # Ajouter les fichiers
        for file in include_files:
            if os.path.exists(file):
                zipf.write(file)
                print(f"✅ Ajout: {file}")
        
        # Ajouter les dossiers
        for dir_name in include_dirs:
            if os.path.exists(dir_name):
                for root, dirs, files in os.walk(dir_name):
                    # Exclure les dossiers inutiles
                    dirs[:] = [d for d in dirs if d not in exclude_dirs]
                    
                    for file in files:
                        if file.endswith(('.pyc', '.pyo', '.so', '.dll')):
                            continue
                        file_path = os.path.join(root, file)
                        arcname = os.path.relpath(file_path, start=os.getcwd())
                        zipf.write(file_path, arcname)
                        print(f"✅ Ajout: {arcname}")
    
    size_mb = os.path.getsize(zip_name) / (1024 * 1024)
    print(f"\n📦 ZIP créé: {zip_name} ({size_mb:.2f} MB)")
    print(f"📁 Nombre de fichiers: {zipf.namelist().__len__()}")
    
    return zip_name

if __name__ == "__main__":
    create_github_package()

# create_github_zip.py
"""
Script pour créer un package ZIP prêt pour GitHub
Exécutez ce script à la racine du projet
"""

import os
import sys
import zipfile
import shutil
import json
from pathlib import Path
from datetime import datetime
import fnmatch

# ========================================
# CONFIGURATION
# ========================================

PROJECT_NAME = "assistant-juridique-ia"
EXCLUDE_PATTERNS = [
    # Dossiers
    "__pycache__",
    ".pytest_cache",
    ".mypy_cache",
    ".ruff_cache",
    "venv",
    ".venv",
    "env",
    ".env",
    "node_modules",
    ".git",
    ".vscode",
    ".idea",
    ".streamlit",
    # Fichiers
    "*.pyc",
    "*.pyo",
    "*.pyd",
    "*.so",
    "*.dylib",
    "*.dll",
    "*.log",
    "*.db",
    "*.sqlite",
    "*.sqlite3",
    "*.zip",
    "*.tar.gz",
    "*.rar",
    "*.7z",
    # Données
    "data/",
    "backups/",
    "uploads/",
    "processed/",
    "chunks/",
    "db_chroma/",
    "invoices/",
    # Credentials
    "credentials/*.json",
    "credentials/*.pickle",
    "*.pem",
    "*.key",
]

# Fichiers obligatoires à inclure
REQUIRED_FILES = [
    "README.md",
    "LICENSE",
    ".env.example",
    ".gitignore",
    "requirements.txt",
    "docker-compose.yml",
]

# Dossiers à inclure
INCLUDE_DIRS = [
    "backend",
    "frontend",
    "email_templates",
    "scripts",
    "docs",
]

# ========================================
# FONCTIONS
# ========================================

def should_exclude(path: Path) -> bool:
    """Vérifie si un fichier/dossier doit être exclu"""
    rel_path = str(path)
    
    # Vérifier les motifs d'exclusion
    for pattern in EXCLUDE_PATTERNS:
        if fnmatch.fnmatch(rel_path, pattern):
            return True
        # Pour les dossiers, vérifier le nom
        if path.is_dir() and pattern in rel_path:
            return True
    return False

def get_project_files(root_dir: Path) -> list:
    """Récupère la liste des fichiers à inclure"""
    files = []
    
    # Ajouter les fichiers obligatoires à la racine
    for file_name in REQUIRED_FILES:
        file_path = root_dir / file_name
        if file_path.exists() and file_path.is_file():
            files.append(file_path)
    
    # Ajouter les dossiers
    for dir_name in INCLUDE_DIRS:
        dir_path = root_dir / dir_name
        if dir_path.exists() and dir_path.is_dir():
            for file_path in dir_path.rglob("*"):
                if file_path.is_file() and not should_exclude(file_path):
                    files.append(file_path)
    
    # Ajouter les fichiers spécifiques à la racine
    for file_pattern in ["*.md", "*.txt", "*.bat", "*.sh", "*.yml", "*.yaml"]:
        for file_path in root_dir.glob(file_pattern):
            if file_path.is_file() and not should_exclude(file_path):
                if file_path not in files:
                    files.append(file_path)
    
    return sorted(files)

def create_zip(root_dir: Path, output_dir: Path = None) -> str:
    """Crée le fichier ZIP"""
    
    if output_dir is None:
        output_dir = root_dir
    
    output_dir = Path(output_dir)
    output_dir.mkdir(parents=True, exist_ok=True)
    
    # Nom du fichier
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    zip_name = f"{PROJECT_NAME}_{timestamp}.zip"
    zip_path = output_dir / zip_name
    
    # Récupérer les fichiers
    files_to_zip = get_project_files(root_dir)
    
    if not files_to_zip:
        print("❌ Aucun fichier trouvé à inclure")
        return None
    
    # Créer le ZIP
    print(f"📦 Création du ZIP: {zip_name}")
    print(f"📁 Dossier source: {root_dir}")
    print(f"📄 {len(files_to_zip)} fichiers à inclure")
    print("=" * 60)
    
    with zipfile.ZipFile(zip_path, 'w', zipfile.ZIP_DEFLATED) as zipf:
        
        for file_path in files_to_zip:
            # Chemin relatif
            rel_path = file_path.relative_to(root_dir)
            arcname = str(rel_path).replace("\\", "/")
            
            # Vérifier la taille (ignorer les fichiers vides)
            if file_path.stat().st_size == 0:
                continue
            
            # Ajouter au ZIP
            zipf.write(file_path, arcname)
            size_kb = file_path.stat().st_size / 1024
            print(f"  ✅ {arcname} ({size_kb:.1f} KB)")
    
    # Statistiques
    size_mb = zip_path.stat().st_size / (1024 * 1024)
    file_count = len(zipf.namelist())
    
    print("=" * 60)
    print(f"✅ ZIP créé avec succès !")
    print(f"📦 Fichier: {zip_path.name}")
    print(f"📊 Taille: {size_mb:.2f} MB")
    print(f"📄 Fichiers inclus: {file_count}")
    print(f"📁 Emplacement: {zip_path}")
    
    return str(zip_path)

def create_zip_with_cleanup(root_dir: Path) -> str:
    """Crée le ZIP après avoir nettoyé le projet"""
    
    print("🧹 Nettoyage préalable...")
    
    # Supprimer les fichiers inutiles
    patterns = ["*.pyc", "*.pyo", ".DS_Store", "Thumbs.db", "*.log"]
    for pattern in patterns:
        for file_path in root_dir.rglob(pattern):
            try:
                file_path.unlink()
                print(f"  🗑️ Supprimé: {file_path}")
            except:
                pass
    
    # Supprimer les dossiers vides
    for dir_path in sorted(root_dir.rglob("*"), reverse=True):
        try:
            if dir_path.is_dir() and not any(dir_path.iterdir()):
                dir_path.rmdir()
                print(f"  🗑️ Dossier vide supprimé: {dir_path}")
        except:
            pass
    
    return create_zip(root_dir)

if __name__ == "__main__":

"""
Vérifie le contenu du ZIP créé
"""

import zipfile
from pathlib import Path

def verify_zip(zip_path: str):
    """Vérifie le contenu du ZIP"""
    
    print(f"🔍 Vérification du ZIP: {zip_path}")
    print("=" * 60)
    
    with zipfile.ZipFile(zip_path, 'r') as zipf:
        files = zipf.namelist()
        
        print(f"📄 Total: {len(files)} fichiers")
        print("\n📁 Structure:")
        
        # Organiser par dossier
        dirs = {}
        for f in files:
            parts = f.split('/')
            if len(parts) > 1:
                dir_name = parts[0]
                if dir_name not in dirs:
                    dirs[dir_name] = []
                dirs[dir_name].append(f)
        
        for dir_name, items in sorted(dirs.items()):
            print(f"\n  📁 {dir_name}/ ({len(items)} fichiers)")
            for item in items[:10]:
                print(f"    📄 {item}")
            if len(items) > 10:
                print(f"    ... et {len(items) - 10} autres")
        
        print("\n" + "=" * 60)
        
        # Vérifier les fichiers essentiels
        essentials = ["README.md", "LICENSE", ".env.example", "requirements.txt"]
        missing = [e for e in essentials if e not in files]
        
        if missing:
            print("⚠️ Fichiers essentiels manquants:")
            for m in missing:
                print(f"  ❌ {m}")
        else:
            print("✅ Tous les fichiers essentiels sont présents")

if __name__ == "__main__":
    # Trouver le dernier ZIP
    zip_dir = Path("outputs")
    if zip_dir.exists():
        zips = list(zip_dir.glob("*.zip"))
        if zips:
            latest = max(zips, key=lambda x: x.stat().st_mtime)
            verify_zip(str(latest))
        else:
            print("❌ Aucun ZIP trouvé")
    else:
        print("❌ Dossier outputs/ non trouvé")
